cmake_minimum_required(VERSION 3.22)
project(SuperformulaReverb VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(JUCE_PATH "" CACHE PATH "Existing JUCE source checkout")
if(JUCE_PATH)
  add_subdirectory("${JUCE_PATH}" juce)
else()
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(JUCE)
endif()
juce_add_binary_data(Brand SOURCES Assets/cva-labs-logo.png)
juce_add_plugin(SuperformulaReverb COMPANY_NAME "CVA Labs" BUNDLE_ID "com.cvalabs.superformulareverb" PLUGIN_MANUFACTURER_CODE Cval PLUGIN_CODE Sfrv FORMATS VST3 PRODUCT_NAME "Superformula Reverb" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(SuperformulaReverb PRIVATE Source/Plugin.cpp)
target_compile_definitions(SuperformulaReverb PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(SuperformulaReverb PRIVATE Brand juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
enable_testing()
add_executable(DspTests Tests/DspTests.cpp)
target_include_directories(DspTests PRIVATE Source)
add_test(NAME GeometryAndReverb COMMAND DspTests)
juce_add_console_app(PluginChecks PRODUCT_NAME "Superformula Plugin Checks")
target_sources(PluginChecks PRIVATE Tests/PluginChecks.cpp)
target_compile_definitions(PluginChecks PRIVATE JUCE_PLUGINHOST_VST3=1)
target_link_libraries(PluginChecks PRIVATE SuperformulaReverb juce::juce_audio_utils)
add_test(NAME PluginStateAndUI COMMAND PluginChecks)
add_test(NAME VST3Load COMMAND PluginChecks "${CMAKE_BINARY_DIR}/SuperformulaReverb_artefacts/$<CONFIG>/VST3/Superformula Reverb.vst3")
